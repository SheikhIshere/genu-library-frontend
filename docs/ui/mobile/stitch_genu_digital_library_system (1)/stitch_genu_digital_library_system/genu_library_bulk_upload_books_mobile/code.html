# Genu Sanctuary PowerShell Folio Dispatch
$headers = @{ "Authorization" = "Bearer $GENU_ARCHIVE_KEY" }
$books = @(
  @{ title = "The Alchemist's Codex"; author = "Alistair Vance"; tags = @("Hermetic", "Alchemy") },
  @{ title = "De Sphaera Mundi"; author = "Johannes Sacrobosco"; tags = @("Astronomy") },
  @{ title = "Iron Foundry Chronicles"; author = "Valerius"; tags = @("Craft") }
)
$books | ForEach-Object {
  Invoke-RestMethod -Uri "https://api.genu.library/v1/folios/bulk" -Method Post -Body ($_ | ConvertTo-Json) -Headers $headers
}
