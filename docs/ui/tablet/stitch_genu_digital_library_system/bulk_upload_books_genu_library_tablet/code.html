# Sanctum Batch Folio Creator
$ApiKey = "sanctum_guild_token_prod_9941"
$Folios = @(
  @{ Title="Codex Arcanum"; Author="M. Vance"; Genre="Alchemy"; File=".\arcana.pdf" },
  @{ Title="The Astrolabe Manual"; Author="E. Wright"; Genre="Astronomy"; File=".\astrolabe.pdf" }
)
foreach ($item in $Folios) {
  Write-Host "Ingesting $($item.Title)..." -ForegroundColor DarkYellow
  Invoke-RestMethod -Uri "https://sanctum.genu.lib/api/v1/folios/bulk" `
    -Method Post -Headers @{ "Authorization" = "Bearer $ApiKey" } `
    -Body ($item | ConvertTo-Json)
}
